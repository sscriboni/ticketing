# elenco servizi completo
sudo systemctl list-units --type=service

# elenco servizi personalizzati
ls -la /etc/systemd/system/*.service

# stato di un servizio
sudo systemctl status web_app_api

# modificare il file config di un servizio
sudo nano /etc/systemd/system/web_app_api.service

# riavviare i servizi rileggendo i file config
sudo systemctl daemon-reload

# riavviare un servizio
sudo systemctl restart web_app_api

curl http://localhost:5002/health
