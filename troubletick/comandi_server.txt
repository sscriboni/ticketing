# elenco servizi ATTIVI
sudo systemctl list-units --type=service --state=running

# elenco servizi completo
sudo systemctl list-units --type=service

# elenco servizi personalizzati
ls -la /etc/systemd/system/*.service

# stato di un servizio
sudo systemctl status web_app_api

# log di un servizio
sudo journalctl -u web_app_api.service -e -x --no-pager

# riavviare un servizio
sudo systemctl restart web_app_api

# modificare il file config di un servizio
sudo nano /etc/systemd/system/web_app_api.service

# riavviare i servizi rileggendo i file config
sudo systemctl daemon-reload

# verificare su server le api di un servizio
curl http://localhost:5002/health

# elenco virtual host apache
sudo httpd -S

# test config apache
sudo httpd -t

# ricarica apache
sudo systemctl reload httpd

# controllare log mail inviate
sudo tail -f /var/log/maillog | grep -iE "status=(bounced|deferred|failed)|reject|error"
